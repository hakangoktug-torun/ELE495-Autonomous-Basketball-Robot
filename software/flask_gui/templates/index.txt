    <script>
        const demoDurationSeconds = 5 * 60;
        const startedAt = Date.now();

        async function postJson(url, body = {}) {
            const response = await fetch(url, {
                method: "POST",
                headers: {"Content-Type": "application/json"},
                body: JSON.stringify(body)
            });
            renderStatus(await response.json());
        }

        async function refreshStatus() {
            try {
                const response = await fetch("/api/status");
                renderStatus(await response.json());
                setConnection(true);
            } catch (error) {
                setConnection(false);
            }
        }

        function renderStatus(status) {
            const readableMode = status.mode.replace("_", " ").toUpperCase();
            document.getElementById("mode").textContent = readableMode;
            document.getElementById("mode-card").textContent = readableMode;
            document.getElementById("score").textContent = status.score;
            document.getElementById("battery").textContent = `${status.battery_percent}%`;
            document.getElementById("pose").textContent =
                `${status.pose.x_cm.toFixed(1)}, ${status.pose.y_cm.toFixed(1)}, ${status.pose.heading_deg.toFixed(0)} deg`;
            document.getElementById("front-clear").textContent = status.front_clear ? "acik" : "engel var";
            document.getElementById("drive-left").textContent = status.drive_left.toFixed(2);
            document.getElementById("drive-right").textContent = status.drive_right.toFixed(2);
            document.getElementById("shooter").textContent =
                status.shooter_armed ? `hazir (${status.shooter_speed.toFixed(2)})` : "kapali";
            updateModeBadge(status.mode, status.front_clear);
            updateMap(status);

            const history = document.getElementById("task-history");
            history.innerHTML = "";
            status.task_history.forEach((item) => {
                const row = document.createElement("li");
                const parts = item.split(" - ");
                const time = document.createElement("span");
                const message = document.createElement("span");
                time.className = "history-time";
                time.textContent = `[${parts[0]}]`;
                message.textContent = parts.slice(1).join(" - ") || item;
                row.append(time, message);
                history.appendChild(row);
            });
        }

        function updateMap(status) {
            const svg = document.getElementById("sim-map");
            if (svg.dataset.sized !== "1") {
                svg.setAttribute("viewBox", `0 0 ${status.court_width_cm} ${status.court_length_cm}`);
                svg.querySelector(".map-court").setAttribute("width", status.court_width_cm);
                svg.querySelector(".map-court").setAttribute("height", status.court_length_cm);
                svg.dataset.sized = "1";
            }

            const marker = document.getElementById("robot-marker");
            const heading = document.getElementById("robot-heading");
            const x = status.pose.x_cm;
            const y = status.pose.y_cm;
            const rad = (status.pose.heading_deg * Math.PI) / 180;
            const arrowLen = Math.max(status.court_width_cm, status.court_length_cm) * 0.08;

            marker.setAttribute("cx", x);
            marker.setAttribute("cy", y);
            heading.setAttribute("x1", x);
            heading.setAttribute("y1", y);
            heading.setAttribute("x2", x + arrowLen * Math.cos(rad));
            heading.setAttribute("y2", y + arrowLen * Math.sin(rad));

            document.getElementById("sim-badge").style.display =
                status.simulated_motion ? "inline-flex" : "none";
        }

        function setConnection(isConnected) {
            const connection = document.getElementById("connection");
            connection.textContent = isConnected ? "Bagli" : "Baglanti yok";
            connection.className = isConnected ? "connection connected" : "connection disconnected";
        }

        function updateModeBadge(mode, frontClear) {
            const badge = document.getElementById("mode-badge");
            badge.className = "mode-badge";
            if (mode === "emergency_stop" || mode === "error") {
                badge.classList.add("mode-danger");
            } else if (!frontClear || mode === "idle") {
                badge.classList.add("mode-waiting");
            } else if (mode === "autonomous") {
                badge.classList.add("mode-auto");
            } else {
                badge.classList.add("mode-ready");
            }
        }

        function updateTimer() {
            const elapsed = Math.floor((Date.now() - startedAt) / 1000);
            const remaining = Math.max(0, demoDurationSeconds - elapsed);
            const minutes = Math.floor(remaining / 60).toString().padStart(2, "0");
            const seconds = (remaining % 60).toString().padStart(2, "0");
            document.getElementById("timer").textContent = `${minutes}:${seconds}`;
        }

        document.querySelectorAll("[data-drive]").forEach((button) => {
            button.addEventListener("click", () => {
                const [left, right] = button.dataset.drive.split(",").map(Number);
                postJson("/api/manual", {left, right});
            });
        });

        document.querySelectorAll("[data-score]").forEach((button) => {
            button.addEventListener("click", () => {
                postJson("/api/score", {points: Number(button.dataset.score)});
            });
        });

        document.getElementById("auto-step").addEventListener("click", () => postJson("/api/autonomous-step"));
        document.getElementById("shoot").addEventListener("click", () => postJson("/api/shoot"));
        document.getElementById("score-reset").addEventListener("click", () => postJson("/api/score/reset"));
        document.getElementById("reset").addEventListener("click", () => postJson("/api/reset"));
        document.getElementById("emergency").addEventListener("click", () => postJson("/api/emergency-stop"));

        refreshStatus();
        updateTimer();
        setInterval(refreshStatus, 700);
        setInterval(updateTimer, 1000);
    </script>
